package stqa_71;

import jxl.*;
import jxl.write.*;
import org.testng.annotations.*;
import java.io.*;

public class EvenOdd_71 {

    @BeforeClass
    public void f1() {
        // Initialization if needed
    }

    @Test
    public void testImportExport() throws Exception {

        String inputPath =
                "C:\\Users\\Ruchit\\OneDrive\\Documents\\ruchit\\ST_71\\EvenOdd.xls";

        String outputPath =
                "C:\\Users\\Ruchit\\OneDrive\\Documents\\ruchit\\ST_71\\EvenOdd_Result.xls";

        
        FileInputStream fi = new FileInputStream(inputPath);

        
        Workbook w = Workbook.getWorkbook(fi);

        Sheet s = w.getSheet(0);

        
        FileOutputStream fo = new FileOutputStream(outputPath);

        WritableWorkbook wwb = Workbook.createWorkbook(fo);

        WritableSheet ws = wwb.createSheet("result", 0);

        int rows = s.getRows();
        int cols = s.getColumns();

        
        for (int i = 0; i < rows; i++) {

            for (int j = 0; j < cols; j++) {

                String content = s.getCell(j, i).getContents();

                ws.addCell(new Label(j, i, content));
            }
        }

        
        int resultCol = cols;

        ws.addCell(new Label(resultCol, 0, "Result"));

        
        for (int i = 1; i < rows; i++) {

            String value = s.getCell(0, i).getContents().trim();

            if (!value.isEmpty()) {

                try {

                    int x = Integer.parseInt(value);

                    String parity = (x % 2 == 0) ? "Even" : "Odd";

                    ws.addCell(new Label(resultCol, i, parity));

                } catch (NumberFormatException e) {

                    ws.addCell(
                            new Label(resultCol, i, "Invalid Number")
                    );
                }
            }
        }

        
        wwb.write();
        wwb.close();

        fo.close();

        
        w.close();
        fi.close();
    }
}
